# STEP SNIPPET: insert after app-token and before scorecard in reusable-standard-ci.yml.
- name: Collect security alert counts
  id: security-alerts
  if: always()
  uses: actions/github-script@v7
  env:
    APP_TOKEN_AVAILABLE: ${{ steps.app-token.outputs.token != '' }}
  with:
    github-token: ${{ steps.app-token.outputs.token || github.token }}
    script: |
      const { owner, repo } = context.repo;
      if (process.env.APP_TOKEN_AVAILABLE !== 'true') {
        core.warning('App token unavailable; attempting alert retrieval with the workflow token.');
      }

      // Only counts and availability leave this step. Never log alert bodies.
      core.setOutput('code_available', 'no');
      core.setOutput('secret_available', 'no');
      core.setOutput('code_critical', '');
      core.setOutput('code_high', '');
      core.setOutput('secret_open', '');
      core.setOutput('code_note', '');
      core.setOutput('secret_note', '');

      try {
        // Repository alert snapshot; not a claim that this PR's analysis finished.
        const alerts = await github.paginate('GET /repos/{owner}/{repo}/code-scanning/alerts', {
          owner, repo, state: 'open', per_page: 100
        });
        const severity = alert => String(alert.rule?.security_severity_level || '').toLowerCase();
        core.setOutput('code_critical', String(alerts.filter(alert => severity(alert) === 'critical').length));
        core.setOutput('code_high', String(alerts.filter(alert => severity(alert) === 'high').length));
        core.setOutput('code_available', 'yes');
        core.info('Code-scanning alert collection succeeded.');
      } catch (error) {
        const status = Number.isInteger(error.status) ? `HTTP ${error.status}` : 'request error';
        const note = `Code-scanning alerts unavailable (${status}); check App access, permissions and feature availability.`;
        core.setOutput('code_note', note);
        core.warning(note);
      }

      try {
        const alerts = await github.paginate('GET /repos/{owner}/{repo}/secret-scanning/alerts', {
          owner, repo, state: 'open', per_page: 100
        });
        core.setOutput('secret_open', String(alerts.length));
        core.setOutput('secret_available', 'yes');
        core.info('Secret-scanning alert collection succeeded.');
      } catch (error) {
        const status = Number.isInteger(error.status) ? `HTTP ${error.status}` : 'request error';
        const note = `Secret-scanning alerts unavailable (${status}); check App access, permissions and feature availability.`;
        core.setOutput('secret_note', note);
        core.warning(note);
      }
