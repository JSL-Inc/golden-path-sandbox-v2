- name: Diagnose security alert API access
  if: always()
  shell: bash
  env:
    GH_TOKEN: ${{ steps.app-token.outputs.token }}
    REPO: ${{ github.repository }}
    APP_ID_CONFIGURED: ${{ inputs.app_client_id != '' }}
    APP_TOKEN_OUTCOME: ${{ steps.app-token.outcome }}
  run: |
    set -uo pipefail

    echo "Repository being queried: $REPO"
    echo "App ID input configured: $APP_ID_CONFIGURED"
    echo "App token creation outcome: $APP_TOKEN_OUTCOME"

    # Deliberately avoid the normal scorecard's github.token fallback.
    if [[ -z "${GH_TOKEN:-}" ]]; then
      echo "::warning::No GitHub App token was produced. Inspect the preceding token-generation step."
      exit 0
    fi

    if ! command -v gh >/dev/null 2>&1; then
      echo "::warning::GitHub CLI is not installed on this runner."
      exit 0
    fi

    for scanner in code-scanning secret-scanning; do
      echo "Checking access to ${scanner} alerts..."
      # --silent discards successful alert payloads; gh still reports API errors.
      # Do not add --verbose, --include, or shell tracing to this diagnostic.
      if gh api --silent \
        -H "Accept: application/vnd.github+json" \
        "/repos/${REPO}/${scanner}/alerts?state=open&per_page=1"; then
        echo "${scanner}: API request succeeded (including when no alerts exist)."
      else
        echo "::warning::${scanner}: API request failed; see the GitHub CLI error above."
      fi
    done
